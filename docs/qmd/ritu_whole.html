Author

Published

April 9, 2026

Ritu cohort — Local/Early stage only. This page repeats the Jiaxuan whole-cohort analysis on the subset of patients with tumor_stage == "Local/Early" (stage I–II / local / neoadjuvant). All 621 advanced/metastatic patients are excluded, leaving N = 322 (189 carboplatin, 133 cisplatin). tumor_stage is dropped from Table 1 and from every adjuster set because it is constant within this cohort. The exposure and the adjuster set are otherwise unchanged from the Jiaxuan project.

Only the long-horizon kidney outcomes are modelled on this page. AKI within 1 year and the CKD composite are carried through Fine-Gray competing-risk models in Section 4, with death at any time as the competing event. The 90-day adverse-event, hospitalization and 90-day-death models this page used to carry have been removed; those outcomes are still derived in Data Management and remain in the saved master, they are simply no longer modelled. Read both models with their event counts in view — 51 AKI and 26 CKD-composite events in 322 patients against ~20 covariate degrees of freedom, so both are over-parameterised. See RITU_PROJECT.md.

.save_pptx <- function(p, path) {
  tryCatch({
    doc <- officer::read_pptx()
    doc <- officer::add_slide(doc, layout = "Title and Content",
                              master = "Office Theme")
    doc <- officer::ph_with(doc, rvg::dml(ggobj = p),

    print(doc, target = path)
    TRUE
  }, error = function(e) FALSE)
}

# fig.process runs in the chunk eval context. For stacked helpers, the helper
# stashes .pptx_pair[[label]] = list(default=p, fine=p_fine) so we can write
# two PPTX files. Otherwise fall back to last_plot() and one file.
knitr::opts_chunk$set(fig.process = function(filename) {
  label <- knitr::opts_current$get("label")
  if (is.null(label)) return(filename)
  pair <- .pptx_pair[[label]]
  if (!is.null(pair) && inherits(pair$default, "ggplot") && inherits(pair$fine, "ggplot")) {
    p1 <- file.path(pptx_dir, paste0(label, ".pptx"))
    p2 <- file.path(pptx_dir, paste0(label, "_10pct.pptx"))
    ok1 <- .save_pptx(pair$default, p1)
    ok2 <- .save_pptx(pair$fine,    p2)
    rels <- c(if (ok1) paste0(pptx_rel_prefix, label, ".pptx"),
              if (ok2) paste0(pptx_rel_prefix, label, "_10pct.pptx"))
    if (length(rels)) .pptx_written[[label]] <- rels
    return(filename)
  }
  p <- tryCatch(ggplot2::last_plot(), error = function(e) NULL)
  if (is.null(p) || !inherits(p, "ggplot")) return(filename)
  pptx_path <- file.path(pptx_dir, paste0(label, ".pptx"))
  if (.save_pptx(p, pptx_path)) {
    .pptx_written[[label]] <- paste0(pptx_rel_prefix, label, ".pptx")
  }
  filename
})

# Restricting to Local/Early disease leaves several outcomes with very few
# events (90-day death = 2; thrombocytopenia grade 3 and the individual
# hospitalization reasons are in the single digits). A model that cannot be
# estimated raises an error, which would abort the entire render, so every
# table / row / figure helper is wrapped below. On success the output is exactly

# of the table or figure. No model specification is changed.
.wrap_tbl <- function(f) {
  force(f)
  function(...) tryCatch(f(...), error = function(e) knitr::asis_output(paste0(
    "\n::: {.callout-warning}\n",
    "## Model not estimable in this cohort\n\n",
    "Too few events among the ", nrow(dat), " Local/Early patients to fit ",
    "this model with the full adjuster set.\n\n",
    "Error: `", conditionMessage(e), "`\n:::\n")))
}
.wrap_row <- function(f) {
  force(f)
  function(time_var, status_var, label, ...) tryCatch(
    f(time_var, status_var, label, ...),
    error = function(e) {
      ev <- tryCatch(
        sum(as.character(dat[[status_var]]) == "1", na.rm = TRUE),
        error = function(e2) NA_integer_)
      tibble::tibble(
        Outcome = label, Model = "not estimable",
        N = NA_integer_, `Excluded (missing cov.)` = NA_integer_,
        Events = as.integer(ev),
        `(sub)HR (95% CI)` = "not estimable", `p-value` = "\u2014")
    })
}
.wrap_plot <- function(f) {
  force(f)
  function(...) tryCatch(f(...), error = function(e)
    ggplot2::ggplot() +

        label = paste0("Model not estimable in this cohort\n",
                       "(too few events among the ", nrow(dat),
                       " Local/Early patients)\n\n",
                       strwrap(conditionMessage(e), width = 60,
                               prefix = "\n", initial = ""))) +
      ggplot2::theme_void())
}

# Table 1 via tableone. Factor-like columns (factor / character / logical and
# indicator columns of 0/1) are summarised as n (%); every other numeric column
# as median [IQR], except the variables named in `normal_vars` (mean (SD)).
# Categorical variables with any observed cell < 5 get Fisher's exact test.
tableone_mat <- function(d, strata = NULL, normal_vars = "age") {
  d     <- as.data.frame(d)
  vars  <- setdiff(names(d), strata)
  is_fl <- vapply(d[vars], function(x)
    is.factor(x) || is.character(x) || is.logical(x) ||
      (is.numeric(x) && all(stats::na.omit(unique(x)) %in% c(0, 1))),
    logical(1))
  fct_vars  <- vars[is_fl]
  cont_vars <- vars[!is_fl]

  if (is.null(strata)) {
    tab   <- tableone::CreateTableOne(vars = vars, data = d,
                                      factorVars = fct_vars, test = FALSE)
    small <- NULL
  } else {
    tab <- tableone::CreateTableOne(
      vars = vars, strata = strata, data = d, factorVars = fct_vars,
      addOverall = TRUE, test = TRUE, argsExact = list(workspace = 2 * 10^7))
    small <- fct_vars[vapply(fct_vars, function(v)
      any(table(d[[v]], d[[strata]]) < 5), logical(1))]
    if (!length(small)) small <- NULL
  }

  mat <- print(tab, nonnormal = setdiff(cont_vars, normal_vars), exact = small,
               showAllLevels = TRUE, missing = TRUE, printToggle = FALSE,
               noSpaces = TRUE, quote = FALSE)
  tibble::as_tibble(cbind(Characteristic = rownames(mat), mat),
                    .name_repair = "unique")
}

# Create eGFR ratio: CKD-EPI Cr-Cys (unindexed) / Cockcroft-Gault
# Also create per-10 mL/min rescaled versions of cockcroft and CKD-EPI for
# reporting (HRs per 1 mL/min are clinically uninterpretable; convention in
# nephro-onc is per 10 mL/min — bio_agent confirmed). Adjusters keep raw scale.
dat <- dat %>%
  mutate(
    # Ratio uses UNCAPPED eGFR
    egfr_ratio    = ckd_epi_gfr_cre_cys_unindex / cockcroft,
    egfr_ratio_per10 = -egfr_ratio * 10,  # per 0.1-unit *decrease* reporting scale
    egfr_discrepancy = ckd_epi_gfr_cre_cys_unindex - cockcroft,
    # per-10-mL/min DECLINE reporting scale for eGFR discrepancy: HR for this
    # variable reads as the multiplicative change in subHR per 10 mL/min lower
    # discrepancy (i.e., when CKD-EPI Cr-Cys is 10 mL/min farther below CG).
    egfr_disc_per10dec = -egfr_discrepancy / 10,
    # Sensitivity-analysis eGFR ratio (Section 11):
    # numerator = CKD-EPI 2012 cystatin C only (baseline_cys_c_egfr_unindexed)
    # denominator = CKD-EPI 2021 race-free creatinine only (baseline_cre_egfr_unindexed)
    # (Both unindexed; the ratio itself is invariant to whether indexed or unindexed
    # because BSA cancels.) per10 below is the per-0.1 DECLINE reporting scale.
    egfr_ratio_sens       = baseline_cys_c_egfr_unindexed / baseline_cre_egfr_unindexed,
    egfr_ratio_sens_per10 = -egfr_ratio_sens * 10,
    # Percent-difference eGFR (new primary exposure):
    # egfr_diff_pct = 100 * (CKD-EPI Cr-Cys unindexed - Cockcroft-Gault) / Cockcroft-Gault
    # egfr_diff_pct_per10: HR reads as multiplicative change per 10 percentage-point DECLINE.
    egfr_diff_pct         = 100 * (ckd_epi_gfr_cre_cys_unindex - cockcroft) / cockcroft,
    egfr_diff_pct_per10   = -egfr_diff_pct / 10,
    # Percent-difference eGFR for sensitivity analysis:
    # numerator - denominator = CKD-EPI 2012 CysC - CKD-EPI 2021 Cr, both unindexed.
    egfr_diff_pct_sens        = 100 * (baseline_cys_c_egfr_unindexed - baseline_cre_egfr_unindexed) / baseline_cre_egfr_unindexed,
    egfr_diff_pct_sens_per10  = -egfr_diff_pct_sens / 10,
    # Binary form of the same exposure: 1 when the cystatin C eGFR sits more
    # than 10 % below the creatinine eGFR. Reference level is ">= -10%".
    egfr_diff_pct_sens_cat    = factor(
      ifelse(egfr_diff_pct_sens < -10, "< -10%", ">= -10%"),
      levels = c(">= -10%", "< -10%")),
    # Stricter cut at -30 %: a much smaller, more clearly discordant group.
    egfr_diff_pct_sens_cat30  = factor(
      ifelse(egfr_diff_pct_sens < -30, "< -30%", ">= -30%"),
      levels = c(">= -30%", "< -30%")),
    # "Survivor": still in follow-up beyond one year.
    survivor_365              = days_last_follow_up > 365,
    cockcroft_per10 = cockcroft / 10,
    ckd_epi_per10   = ckd_epi_gfr_cre_cys_unindex / 10
  )

days_last_follow_up = last_follow_up_date − baseline_sample_date. Two alternative definitions of “follow-up time” are shown side-by-side so the reader can pick the most defensible one for their purpose:

2.2 Long-Horizon Kidney Outcomes (CKD / ESKD / AKI)

Ported from the ICI-CKD project and built in Sections 5.16–5.19 of Data Management. Creatinine for these comes from the Ritu_project RPDR pull (Oct 2026).

None of these are censored at 90 days. A qualifying CKD episode has to be sustained for ≥ 90 days, so they run over the full length of RPDR creatinine follow-up. aki is restricted to events within 1 year. This section reports all of them descriptively; aki and ckd_composite are the two that are modelled — see the Fine-Gray section below. The rest are descriptive only.

Definitions as of the latest revision. aki is now restricted to events within 1 year of baseline (aki_any keeps the unrestricted count for reference and is not modelled). ckd_incidence uses a > 25 % eGFR drop; ckd_incidence_any drops the percent requirement entirely; ckd_progression now additionally requires a > 30 % further drop, where it previously counted any patient who was simply below 60 before and after — which flagged stable CKD as progression and is why that count fell sharply.

aki is also not the same variable as AKI_g2_90days / AKI_kdigo_90days: it compares each creatinine against the patient’s own preceding 90-day median (or 2-day low) rather than against a fixed pre-baseline value. The cross-tabulation below shows how far the two diverge.

2.3 Delta eGFR (creatinine) vs Baseline at 90 / 180 / 365 Days

Change in creatinine eGFR from eGFR_CRE_baseline, using the creatinine closest to each mark and taking it only when it falls within ± 45 days of that mark. Days run from baseline_sample_date. Built in Section 5.16b of Data Management.

Negative means decline. delta_egfr_d90 is eGFR at day 90 minus eGFR at baseline, so a fall in kidney function is a negative number — the opposite sign to pct_drop in the data-management pipeline.

The ± 45-day window matches the window used for every other baseline lab in this pipeline (pre_CRE_45days), and it is the widest window that stops the 90-day and 180-day marks drawing on the same measurement. It costs coverage at the later marks — the n column below shows how much.

4 Fine-Gray Models — eGFR Difference as (CysC − Cr) / Cr, adjusted for CKD-EPI 2021 creatinine eGFR (unindexed, mL/min)

These are the only two outcomes modelled on this page. Both run on the long-horizon kidney outcomes built in Sections 5.16–5.19 of Data Management: AKI is administratively censored at 365 days, the CKD composite runs to the last creatinine (day 0 → last follow-up). Death at any time is the competing event in both, taken from death_days_any rather than the 90-day death_90 flag, which cannot place a death in time beyond day 90. death_days_any is the union of the two sources that carry a death date — the RPDR Date_Of_Death when there is one, otherwise the REDCap last-follow-up date for a patient REDCap records as dead. Someone flagged deceased in RPDR with no date who is still alive in REDCap stays censored.

The exposure in both sections below is egfr_diff_pct_sens_per10 — 100 × (eGFR_CysC − eGFR_Cr) / eGFR_Cr, negated and divided by 10, so every subdistribution hazard ratio reads per 10 percentage-point decline. Cystatin C against creatinine alone, both unindexed. The two sections differ only in the kidney adjuster: creatinine eGFR in absolute mL/min here, indexed to body surface area in the next. All other adjusters are the project’s standard set (Section 1).

Each table shows univariable (one variable at a time) and multivariable Fine-Gray subdistribution hazard ratios side-by-side. A Word version of each table is saved during render — the download link sits under its heading.

.adj_set <- function(pred_var, pre_labs, egfr_adj) {
  adj <- c("age","sex","score","ecog_score_cat2",
           "cancer_type_cat_7","platin_group","high_dose",
           "Paclitaxel","Pemetrexed","Gemcitabine","ici",
           "bmi","smoking","thyroid")
  # Drop platin_group when predictor requires dose data (carbo-only after drop_na)
  if (grepl("dose_discrep", pred_var)) adj <- setdiff(adj, "platin_group")
  if (egfr_adj == "cockcroft") adj <- c(adj, "cockcroft")
  if (egfr_adj == "ckd_epi")   adj <- c(adj, "ckd_epi_gfr_cre_cys_unindex")
  # Indexed to BSA (mL/min/1.73m2), as opposed to the absolute mL/min above.
  if (egfr_adj == "ckd_epi_idx") adj <- c(adj, "ckd_epi_gfr_cre_cys")
  # Creatinine-only CKD-EPI 2021, race-free. Unlike the Cr-Cys forms above
  # it shares no measurement with the exposure.
  if (egfr_adj == "cre")     adj <- c(adj, "baseline_cre_egfr_unindexed")
  if (egfr_adj == "cre_idx") adj <- c(adj, "baseline_cre_egfr")
  if (!is.null(pre_labs)) adj <- c(adj, pre_labs)
  adj
}

# fg_model: returns univariable + multivariable merged Fine-Gray table.
# If docx_name is supplied, also saves the table as a .docx in docx_dir.
fg_model <- function(time_var, status_var, pred_var, pre_labs = NULL,
                     egfr_adj = c("none", "cockcroft", "ckd_epi", "ckd_epi_idx",
                                  "cre", "cre_idx"),
                     docx_name = NULL) {
  egfr_adj <- match.arg(egfr_adj)
  adj <- .adj_set(pred_var, pre_labs, egfr_adj)
  all_vars <- c(pred_var, adj)

  # Multivariable
  fmla_mv <- as.formula(paste0(
    "Surv(", time_var, ", ", status_var, ") ~ ",
    paste(all_vars, collapse = " + ")
  ))
  tbl_mv <- tidycmprsk::crr(fmla_mv, failcode = 1, data = dat) %>%
    tbl_regression(exp = TRUE,
                   pvalue_fun = function(x) style_pvalue(x, digits = 3)) %>%
    add_n()

  # Univariable: one Fine-Gray fit per variable
  y_call <- str2lang(sprintf("Surv(%s, %s)", time_var, status_var))
  tbl_uv <- rlang::inject(
    tbl_uvregression(
      data = dat,
      method = tidycmprsk::crr,
      y = !!y_call,
      include = dplyr::all_of(all_vars),
      exponentiate = TRUE,
      pvalue_fun = function(x) style_pvalue(x, digits = 3),
      method.args = list(failcode = 1)
    )
  )

  out <- tbl_merge(
    tbls = list(tbl_uv, tbl_mv),
    tab_spanner = c("**Univariable**", "**Multivariable**")
  )

  if (!is.null(docx_name)) save_tbl_docx(out, docx_name)
  out
}

fg_model <- .wrap_tbl(fg_model)

7 Binary eGFR Difference (sens) — AKI and CKD Composite

The same two outcomes and the same adjusters as the Fine-Gray sections above, with the exposure entered as a binary split rather than per 10 percentage-point decline.

  • egfr_diff_pct_sens_cat splits at − 10 % — < -10% means the cystatin C eGFR sits more than 10 % below the creatinine eGFR; >= -10% is the reference.
  • egfr_diff_pct_sens_cat30 splits at − 30 %, a smaller and more clearly discordant group.

Each subdistribution hazard ratio is therefore for the exposed group against its reference group, not per unit. Death at any time is the competing event and the kidney adjuster is baseline_cre_egfr (indexed, mL/min/1.73m²), as in the sections above.

plot_cg_vs_epi <- function(highlight_var, threshold, label) {
  d <- dat %>%
    filter(!is.na(cockcroft), !is.na(ckd_epi_gfr_cre_cys_unindex),
           !is.na(.data[[highlight_var]])) %>%
    mutate(highlight = .data[[highlight_var]] >= threshold)

  n_total <- nrow(d)
  n_red   <- sum(d$highlight)

  # Fit two separate OLS models so we can annotate each with its equation + R².
  fit_true  <- lm(ckd_epi_gfr_cre_cys_unindex ~ cockcroft,
                  data = d %>% filter(highlight))
  fit_false <- lm(ckd_epi_gfr_cre_cys_unindex ~ cockcroft,
                  data = d %>% filter(!highlight))
  eq_str <- function(fit) {
    b  <- coef(fit)
    r2 <- summary(fit)$r.squared
    sprintf("y = %.2f x %+.1f   (R² = %.2f)", b[2], b[1], r2)
  }

  ggplot(d, aes(x = cockcroft, y = ckd_epi_gfr_cre_cys_unindex,
                color = highlight, fill = highlight)) +
    geom_point(alpha = 0.6, size = 1.6) +
    # One OLS regression line per level of `highlight` (colour + CI ribbon
    # match the dot colour).
    geom_smooth(method = "lm", formula = y ~ x, se = TRUE,
                alpha = 0.15, linewidth = 1) +
    annotate("text", x = 5, y = 195, hjust = 0, size = 3.4,
             color = "#E31A1C", label = paste0(">= threshold: ", eq_str(fit_true))) +
    annotate("text", x = 5, y = 182, hjust = 0, size = 3.4,
             color = "grey40",  label = paste0("< threshold:  ", eq_str(fit_false))) +
    scale_color_manual(values = c("FALSE" = "grey60", "TRUE" = "#E31A1C"),
                       guide = "none") +
    scale_fill_manual(values = c("FALSE" = "grey60", "TRUE" = "#E31A1C"),
                      guide = "none") +
    coord_cartesian(xlim = c(0, 200), ylim = c(0, 200)) +
    labs(x = "Cockcroft-Gault eGFR (mL/min)",
         y = "CKD-EPI Cr-Cys eGFR, unindexed (mL/min)",
         title = sprintf("%s   |   N = %d, n_red = %d",
                         label, n_total, n_red)) +
    theme_minimal(base_size = 11)
}

plot_ratio_by_group <- function(highlight_var, threshold, label,
                                highlight_label, reference_label) {
  d <- dat %>%
    dplyr::filter(!is.na(egfr_diff_pct), !is.na(.data[[highlight_var]])) %>%
    dplyr::mutate(grp = factor(
      ifelse(.data[[highlight_var]] >= threshold, highlight_label, reference_label),
      levels = c(highlight_label, reference_label)))

  med_tbl <- d %>%
    dplyr::group_by(grp) %>%
    dplyr::summarise(m = median(egfr_diff_pct, na.rm = TRUE),
                     n = dplyr::n(), .groups = "drop")

  pval <- tryCatch(
    wilcox.test(egfr_diff_pct ~ grp, data = d)$p.value,
    error = function(e) NA_real_)
  p_str <- if (is.na(pval)) "NA" else
    ifelse(pval < 0.001, "<0.001", sprintf("%.3f", pval))

  col_vec <- setNames(c("#E31A1C", "grey50"),
                      c(highlight_label, reference_label))

  ggplot(d, aes(x = egfr_diff_pct, fill = grp, color = grp)) +
    geom_density(alpha = 0.35, linewidth = 0.9) +
    geom_vline(xintercept = 0, linetype = "dashed", color = "grey30") +
    geom_vline(data = med_tbl, aes(xintercept = m, color = grp),
               linetype = "dotted", linewidth = 0.8, show.legend = FALSE) +
    scale_fill_manual(values = col_vec) +
    scale_color_manual(values = col_vec) +
    labs(x = "eGFR difference (%): 100 × (CKD-EPI Cr-Cys − CG) / CG",
         y = "Density",
         fill = NULL, color = NULL,
         title = label,
         subtitle = sprintf(
           "%s: N = %d, median = %.1f %%   |   %s: N = %d, median = %.1f %%   |   Mann-Whitney U p = %s",
           highlight_label,
           med_tbl$n[med_tbl$grp == highlight_label],
           med_tbl$m[med_tbl$grp == highlight_label],
           reference_label,
           med_tbl$n[med_tbl$grp == reference_label],
           med_tbl$m[med_tbl$grp == reference_label],
           p_str)) +
    theme_minimal(base_size = 12) +
    theme(legend.position = "top",
          plot.subtitle = element_text(size = 9, color = "grey25"))
}
